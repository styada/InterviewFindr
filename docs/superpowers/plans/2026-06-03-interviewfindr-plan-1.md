# InterviewFindr Plan 1 — Foundation + Vertical Slice

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Ship a self-hosted web app that ingests resumes, fetches jobs from Greenhouse, runs the 6-stage matching/tailoring pipeline, and presents family members with top matches + tailored resume + cover letter for assisted apply. No real auto-apply yet (Plan 2). No Tier 2/3 sources yet (Plan 3).

**Architecture:** FastAPI monolith + Postgres 16 + MinIO + Caddy in Docker Compose on a Hostinger VPS. HTMX + Tailwind + Alpine.js for the UI. APScheduler in-process for daily runs. LLM calls go through a single `LLMProvider` interface; OpenCode Zen is the only concrete provider in this plan.

**Tech Stack:** Python 3.12, FastAPI, SQLAlchemy 2.x, Alembic, Postgres 16, MinIO, Caddy 2, HTMX 2, Tailwind 3 (compiled via standalone CLI), Alpine.js 3, APScheduler 3, httpx, WeasyPrint, python-docx, pdfplumber, pydantic v2, argon2-cffi, pytest, ruff, mypy.

**Spec:** [`docs/superpowers/specs/2026-06-03-interviewfindr-design.md`](../specs/2026-06-03-interviewfindr-design.md)

**Auth decision (locked):** Username + password, argon2id hash, session cookies, no email infrastructure in v1.

```python
from collections.abc import Iterator

from sqlalchemy import create_engine
from sqlalchemy.orm import Session, sessionmaker

from app.core.config import get_settings


def _engine():
    return create_engine(get_settings().database_url, future=True, pool_pre_ping=True)


_engine = _engine()
SessionLocal = sessionmaker(bind=_engine, autoflush=False, expire_on_commit=False)


def get_db() -> Iterator[Session]:
    db = SessionLocal()
    try:
        yield db
    finally:
        db.close()
```

```python
from sqlalchemy.orm import Mapped, mapped_column, relationship

from app.db.base import Base


def _uuid() -> uuid.UUID:
    return uuid.uuid4()


class User(Base):
    __tablename__ = "users"
    id: Mapped[uuid.UUID] = mapped_column(UUID(as_uuid=True), primary_key=True, default=_uuid)

    password_hash: Mapped[Optional[str]] = mapped_column(Text, nullable=True)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())

    profile: Mapped[Optional["Profile"]] = relationship(back_populates="user", uselist=False)


class Profile(Base):
    __tablename__ = "profiles"
    user_id: Mapped[uuid.UUID] = mapped_column(
        UUID(as_uuid=True), ForeignKey("users.id", ondelete="CASCADE"), primary_key=True

    auto_apply_enabled: Mapped[bool] = mapped_column(Boolean, default=False)
    auto_apply_allowlist: Mapped[list[str]] = mapped_column(ARRAY(String), default=list)
    tailoring_thresholds: Mapped[dict] = mapped_column(JSONB, default=dict)
    created_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now())
    updated_at: Mapped[datetime] = mapped_column(DateTime(timezone=True), server_default=func.now(), onupdate=func.now())

    user: Mapped[User] = relationship(back_populates="profile")
```

```python
import uuid

from argon2 import PasswordHasher
from itsdangerous import BadSignature, URLSafeSerializer

from app.core.config import get_settings

_hasher = PasswordHasher()


def hash_password(plaintext: str) -> str:
    return _hasher.hash(plaintext)


def verify_password(plaintext: str, hashed: str) -> bool:
    try:
        return _hasher.verify(hashed, plaintext)
    except Exception:
        return False


def create_session_token(user_id: uuid.UUID | str) -> str:
    s = URLSafeSerializer(get_settings().app_secret_key, salt="session")
    return s.dumps(str(user_id))


def verify_session_token(token: str) -> str | None:
    s = URLSafeSerializer(get_settings().app_secret_key, salt="session")
    try:
        return s.loads(token)
    except BadSignature:
        return None
```

```python
from typing import Annotated

from fastapi import APIRouter, Depends, Form, HTTPException, Response, status
from sqlalchemy import select
from sqlalchemy.orm import Session

from app.api.deps import current_user
from app.core.config import get_settings
from app.core.security import create_session_token, hash_password, verify_password
from app.db.models import User
from app.db.session import get_db
from app.schemas.user import UserOut

router = APIRouter()


@router.post("/auth/register", response_model=UserOut)
def register(
    email: Annotated[str, Form()],
    display_name: Annotated[str, Form()],
    password: Annotated[str, Form(min_length=12)],
    role: Annotated[str, Form()] = "member",
    db: Session = Depends(get_db),
) -> User:
    if db.scalar(select(User).where(User.email == email)):
        raise HTTPException(status.HTTP_409_CONFLICT, "email already registered")
    user = User(
        email=email,
        display_name=display_name,
        role=role,
        password_hash=hash_password(password),
    )
    db.add(user)
    db.commit()
    db.refresh(user)
    return user


@router.post("/auth/login")
def login(
    response: Response,
    email: Annotated[str, Form()],
    password: Annotated[str, Form()],
    db: Session = Depends(get_db),
) -> dict[str, str]:
    user = db.scalar(select(User).where(User.email == email))
    if not user or not user.password_hash or not verify_password(password, user.password_hash):
        raise HTTPException(status.HTTP_401_UNAUTHORIZED, "invalid credentials")
    token = create_session_token(user.id)
    response.set_cookie(
        get_settings().session_cookie_name,
        token,
        httponly=True,
        samesite="lax",
        secure=get_settings().app_env == "production",
        max_age=get_settings().session_ttl_days * 86400,
    )
    return {"status": "ok"}


@router.post("/auth/logout")
def logout(response: Response) -> dict[str, str]:
    response.delete_cookie(get_settings().session_cookie_name)
    return {"status": "ok"}


@router.get("/auth/me", response_model=UserOut)
def me(user: User = Depends(current_user)) -> User:
    return user
```

```python
import httpx

from app.core.config import get_settings
from app.llm.base import LLMResponse


class OpenCodeZenProvider:
    """OpenAI-compatible /v1/chat/completions endpoint."""

    def __init__(self, base_url: str | None = None, api_key: str | None = None) -> None:
        s = get_settings()
        self._base = (base_url or s.opencode_zen_base_url).rstrip("/")
        self._key = api_key or s.opencode_zen_api_key

    async def complete(
        self,
        messages: list[dict],
        *,
        model: str,
        max_tokens: int = 1024,
        temperature: float = 0.2,
        json_mode: bool = False,
    ) -> LLMResponse:
        payload: dict = {
            "model": model,
            "messages": messages,
            "max_tokens": max_tokens,
            "temperature": temperature,
        }
        if json_mode:
            payload["response_format"] = {"type": "json_object"}
        async with httpx.AsyncClient(timeout=60) as client:
            r = await client.post(
                f"{self._base}/chat/completions",
                json=payload,
                headers={"Authorization": f"Bearer {self._key}"},
            )
            r.raise_for_status()
            data = r.json()
        choice = data["choices"][0]
        usage = data.get("usage", {})
        text = choice["message"]["content"]
        in_t = int(usage.get("prompt_tokens", 0))
        out_t = int(usage.get("completion_tokens", 0))
        # Cost is provider-agnostic placeholder; the router will recompute from the
        # configured price table. Default to 0 and let the caller apply pricing.
        return LLMResponse(text=text, input_tokens=in_t, output_tokens=out_t, cost_usd=0.0, raw=data)
```

- [ ] **Step 3: Create `app/llm/router.py`** with model selection + caching

```python
import hashlib
import json
from collections.abc import Awaitable
from datetime import datetime, timedelta

from app.llm.base import LLMProvider, LLMResponse


# Per-token USD pricing (input, output). Tune as OpenCode Zen updates.
PRICES: dict[str, tuple[float, float]] = {
    "kimi-k2.6": (0.000_000_6, 0.000_003),
    "deepseek-v4-pro": (0.000_001, 0.000_005),
    "glm-5.1": (0.000_000_8, 0.000_004),
    "qwen3.7-max": (0.000_001, 0.000_004_5),
}


class _Cache:
    def __init__(self, ttl_seconds: int = 86400) -> None:
        self._store: dict[str, tuple[datetime, LLMResponse]] = {}
        self._ttl = timedelta(seconds=ttl_seconds)

    def get(self, key: str) -> LLMResponse | None:
        if key in self._store:
            ts, val = self._store[key]
            if datetime.utcnow() - ts < self._ttl:
                return val
            del self._store[key]
        return None

    def set(self, key: str, value: LLMResponse) -> None:
        self._store[key] = (datetime.utcnow(), value)


class LLMRouter:
    def __init__(self, providers: dict[str, LLMProvider], cache: _Cache | None = None) -> None:
        self._providers = providers
        self._cache = cache or _Cache()

    def _key(self, provider: str, model: str, messages: list[dict], **kwargs: object) -> str:
        h = hashlib.sha256()
        h.update(provider.encode())
        h.update(b"|")
        h.update(model.encode())
        h.update(b"|")
        h.update(json.dumps(messages, sort_keys=True).encode())
        h.update(b"|")
        h.update(json.dumps(kwargs, sort_keys=True).encode())
        return h.hexdigest()

    async def complete(
        self,
        provider: str,
        model: str,
        messages: list[dict],
        **kwargs: object,
    ) -> LLMResponse:
        key = self._key(provider, model, messages, **kwargs)
        cached = self._cache.get(key)
        if cached is not None:
            return cached
        raw = await self._providers[provider].complete(messages, model=model, **kwargs)
        ip, op = PRICES.get(model, (0.0, 0.0))
        cost = raw.input_tokens * ip + raw.output_tokens * op
        priced = LLMResponse(
            text=raw.text, input_tokens=raw.input_tokens,
            output_tokens=raw.output_tokens, cost_usd=cost, raw=raw.raw,
        )
        self._cache.set(key, priced)
        return priced
```

```python
from io import BytesIO
from pathlib import Path

import pdfplumber
from docx import Document


def extract_text(data: bytes, filename: str) -> str:
    suffix = Path(filename).suffix.lower()
    if suffix == ".pdf":
        return _extract_pdf(data)
    if suffix == ".docx":
        return _extract_docx(data)
    if suffix == ".txt":
        return data.decode("utf-8", errors="replace")
    raise ValueError(f"unsupported resume format: {suffix}")


def _extract_pdf(data: bytes) -> str:
    text = []
    with pdfplumber.open(BytesIO(data)) as pdf:
        for page in pdf.pages:
            t = page.extract_text()
            if t:
                text.append(t)
    return "\n".join(text)


def _extract_docx(data: bytes) -> str:
    doc = Document(BytesIO(data))
    return "\n".join(p.text for p in doc.paragraphs)
```

```python
import re
from dataclasses import dataclass

RED_FLAGS = [
    r"\brockstar\b", r"\bninja\b", r"\bguru\b", r"\bwear many hats\b",
    r"\bfast-paced\b", r"\bcompetitive salary\b",
]
POSITIVE_SIGNALS = [
    r"\$\d{2,3}[,._]?\d{3}",                # $XX,XXX or similar
    r"\d+\s*-\s*\$\d", r"\$\d+\s*-\s*\$?\d",  # ranges
    r"\bbenefits\b", r"\b401k\b", r"\bhealth insurance\b",
    r"\byou will\b", r"\bresponsibilities include\b",
]


@dataclass
class QualityResult:
    score: float  # 0..1
    red_flags: list[str]
    positives: list[str]


def score_jd(text: str) -> QualityResult:
    if not text or len(text) < 200:
        return QualityResult(score=0.0, red_flags=["too_short"], positives=[])
    text_l = text.lower()
    red = [p for p in RED_FLAGS if re.search(p, text_l)]
    pos = [p for p in POSITIVE_SIGNALS if re.search(p, text_l)]
    score = 0.5 + 0.1 * len(pos) - 0.15 * len(red)
    score = max(0.0, min(1.0, score))
    return QualityResult(score=score, red_flags=red, positives=pos)
```

```python
import re
from dataclasses import dataclass

# Hand-curated equivalence map (Plan 3 will add more / load from DB).
TRANSFER_MAP: dict[str, list[tuple[str, float]]] = {
    "HubSpot": [("Salesforce", 0.4), ("CRM", 0.3)],
    "PostgreSQL": [("MySQL", 0.6), ("SQL", 0.8)],
    "Postgres": [("MySQL", 0.6), ("SQL", 0.8)],
    "React": [("Vue", 0.4), ("Frontend", 0.6)],
    "Excel": [("Google Sheets", 0.7), ("Financial modeling", 0.5)],
    "Treasury operations": [("Cash management", 0.7), ("Banking", 0.6)],
    "Commercial banking": [("Banking", 0.8), ("Credit analysis", 0.7)],
}


@dataclass
class SkillMatch:
    coverage: float         # 0..1
    uncovered: list[str]    # must_have strings without direct or transferred match


def _normalize(s: str) -> str:
    return re.sub(r"\s+", " ", s.strip().lower())


def match_must_haves(must_haves: list[str], resume_skills: list[dict]) -> SkillMatch:
    if not must_haves:
        return SkillMatch(coverage=0.0, uncovered=[])
    resume_keys = {_normalize(s["name"]) for s in resume_skills}
    total = 0.0
    uncovered: list[str] = []
    for mh in must_haves:
        mh_n = _normalize(mh)
        if any(rk in mh_n or mh_n in rk for rk in resume_keys):
            total += 1.0
            continue
        transfer = TRANSFER_MAP.get(mh) or TRANSFER_MAP.get(mh_n) or []
        score = 0.0
        for target, conf in transfer:
            if _normalize(target) in resume_keys:
                score = max(score, conf)
        if score > 0:
            total += score
        else:
            uncovered.append(mh)
    return SkillMatch(coverage=total / len(must_haves), uncovered=uncovered)
```

```python
from dataclasses import dataclass

DEFAULT_WEIGHTS = {
    "w_match_coverage": 0.30,
    "w_jd_quality": 0.20,
    "w_seniority_match": 0.20,
    "w_comp_match": 0.10,
    "w_recency": 0.10,
    "w_source_weight": 0.05,
    "w_competition": 0.05,
}

SOURCE_WEIGHTS = {
    "greenhouse": 1.0,
    "lever": 0.9,
    "ashby": 0.9,
    "adzuna": 0.8,
    "jsearch": 0.6,
    "usajobs": 1.0,
}


@dataclass
class ProbabilityInput:
    must_have_cov: float
    jd_quality: float
    seniority_match: float
    comp_match: float
    recency_score: float
    source: str
    competition_density: float = 0.5  # 0 = none, 1 = max


def probability(inp: ProbabilityInput, weights: dict | None = None) -> float:
    w = {**DEFAULT_WEIGHTS, **(weights or {})}
    sw = SOURCE_WEIGHTS.get(inp.source, 0.7)
    p = (
        w["w_match_coverage"] * inp.must_have_cov
        + w["w_jd_quality"] * inp.jd_quality
        + w["w_seniority_match"] * inp.seniority_match
        + w["w_comp_match"] * inp.comp_match
        + w["w_recency"] * inp.recency_score
        + w["w_source_weight"] * sw
        - w["w_competition"] * inp.competition_density
    )
    return max(0.0, min(1.0, p))


def seniority_match(level: str, target: str) -> float:
    """Crude mapping. Real impl in Plan 2 will load a leveled matrix."""
    ladder = ["entry", "IC3", "IC4", "manager", "principal", "director", "vp"]
    if level not in ladder or target not in ladder:
        return 0.5
    diff = abs(ladder.index(level) - ladder.index(target))
    if diff == 0:
        return 1.0
    if diff == 1:
        return 0.7
    return 0.3


def recency_score(posted_at_iso: str | None) -> float:
    if not posted_at_iso:
        return 0.5
    import datetime as dt
    try:
        posted = dt.datetime.fromisoformat(posted_at_iso.replace("Z", "+00:00"))
    except ValueError:
        return 0.5
    now = dt.datetime.now(dt.timezone.utc)
    days = (now - posted).total_seconds() / 86400
    if days < 0:
        days = 0
    if days >= 30:
        return 0.3
    return 1.0 - (days / 30) * 0.7
```

```python
import json
from dataclasses import dataclass

from app.llm.router import LLMRouter

RESUME_PROMPT = """\
You are tailoring a resume for a specific job. Given the candidate's structured
resume and the parsed job, produce a tailored resume JSON matching the candidate's
schema. Mirror the must_have keywords verbatim in the bullets where they don't
misrepresent experience. Reorder bullets to emphasize relevant experience.

CANDIDATE RESUME (JSON):
{resume}

JOB (JSON):
{job}

Return ONLY the tailored resume JSON.
"""

COVER_LETTER_PROMPT = """\
Write a 3-paragraph cover letter for this candidate applying to this job.
Paragraph 1: specific opener referencing the company/role/team, not generic.
Paragraph 2: 2-3 concrete examples from the resume that map to the must-haves, with quantification.
Paragraph 3: specific closer about why this company/team, not boilerplate.

CANDIDATE (JSON): {resume}
JOB (JSON): {job}
"""


@dataclass
class TailoringResult:
    tailored_resume: dict
    cover_letter: str
    model_used: str
    cost_usd: float


async def tailor(
    router: LLMRouter,
    resume: dict,
    job_parsed: dict,
    *,
    model: str = "kimi-k2.6",
) -> TailoringResult:
    resume_resp = await router.complete(
        "opencode_zen", model,
        [{"role": "user", "content": RESUME_PROMPT.format(
            resume=json.dumps(resume, indent=2),
            job=json.dumps(job_parsed, indent=2),
        )}],
        max_tokens=3000, temperature=0.2, json_mode=True,
    )
    tailored = json.loads(resume_resp.text)

    cl_resp = await router.complete(
        "opencode_zen", model,
        [{"role": "user", "content": COVER_LETTER_PROMPT.format(
            resume=json.dumps(resume, indent=2),
            job=json.dumps(job_parsed, indent=2),
        )}],
        max_tokens=900, temperature=0.5,
    )
    return TailoringResult(
        tailored_resume=tailored,
        cover_letter=cl_resp.text,
        model_used=model,
        cost_usd=resume_resp.cost_usd + cl_resp.cost_usd,
    )
```

```python
from dataclasses import dataclass


@dataclass
class ValidationResult:
    ok: bool
    issues: list[str]


def validate_tailored(
    *,
    resume_text: str,
    must_haves: list[str],
    cover_letter: str,
) -> ValidationResult:
    issues: list[str] = []
    if not resume_text or len(resume_text) < 200:
        issues.append("resume_too_short")
    if not cover_letter or len(cover_letter) < 200:
        issues.append("cover_letter_too_short")
    resume_lower = resume_text.lower()
    for mh in must_haves:
        mh_l = mh.lower()
        # accept either the full must_have or its key token (first 3 words)
        token = " ".join(mh_l.split()[:3])
        if token and token not in resume_lower and mh_l not in resume_lower:
            issues.append(f"missing_must_have:{mh}")
    if len(issues) > 2:
        return ValidationResult(ok=False, issues=issues)
    return ValidationResult(ok=len(issues) == 0, issues=issues)
```

```python
    paragraphs = [p.strip() for p in text.split("\n\n") if p.strip()]

def resume_to_plain_text(resume: dict) -> str:
    """For Stage 6 keyword validation."""
    out = []

    for r in resume.get("experience", []):

        out.extend(r.get("bullets", []))
    for s in resume.get("skills", []):
        out.append(s.get("name", ""))
    return "\n".join(out)
```

```python
import uuid
from dataclasses import dataclass

from sqlalchemy import select

from app.db.models import Job, Match, Profile, TailoredArtifact, User

from app.pipeline.stage2_parse import parse_jd

@dataclass
class PipelineConfig:
    jd_quality_min: float = 0.40
    must_have_coverage_min: float = 0.70
    top_n: int = 25


async def upsert_job(db: Session, raw: RawJob) -> Job:
    stmt = pg_insert(Job).values(
        source=raw.source, external_id=raw.external_id, url=raw.url,
        company=raw.company, title=raw.title, location=raw.location,
        remote_type=raw.remote_type, salary_min=raw.salary_min,
        salary_max=raw.salary_max, currency=raw.currency,
        description_html=raw.description_html, description_text=raw.description_text,
        posted_at=raw.posted_at, raw_payload=raw.raw_payload,
    ).on_conflict_do_update(
        index_elements=["source", "external_id"],
        set_={"last_seen_at": Job.last_seen_at},
    ).returning(Job.id)
    return db.execute(stmt).scalar_one()


async def run_for_user(
    db: Session, user: User, profile: Profile, raw_jobs: list[RawJob],
    router: LLMRouter, cfg: PipelineConfig | None = None,
) -> list[Match]:
    """Run the full pipeline for one user over the list of new raw jobs."""
    cfg = cfg or PipelineConfig()
    surviving: list[tuple[Match, dict, dict]] = []  # (match, parsed_jd, tailoring)
    resume = (profile.resume_parsed_json or {})

    for raw in raw_jobs:
        # Stage 1
        q = score_jd(raw.description_text or "")
        if q.score < cfg.jd_quality_min:
            continue

        # Upsert the job first
        job = await upsert_job(db, raw)

        # Stage 2
        parsed = await parse_jd(router, raw)

        # Stage 3
        sm = match_must_haves(parsed.get("must_haves", []), resume.get("skills", []))
        if sm.coverage < cfg.must_have_coverage_min:
            continue

        # Stage 4
        sen = seniority_match(_candidate_level(resume), parsed.get("level", "unknown"))
        rs = recency_score(raw.posted_at)
        comp_match = _comp_match(parsed.get("compensation"), profile.preferences)
        prob_inp = ProbabilityInput(
            must_have_cov=sm.coverage, jd_quality=q.score, seniority_match=sen,
            comp_match=comp_match, recency_score=rs, source=raw.source,
        )
        prob = probability(prob_inp)

        # Persist the Match row
        match = _upsert_match(db, user.id, job.id, prob, q.score, sen, sm.coverage, comp_match)
        surviving.append((match, parsed, raw))

    # Sort, take top-N
    surviving.sort(key=lambda t: t[0].match_score, reverse=True)
    surviving = surviving[: cfg.top_n]

    # Stage 5 + 6
    for match, parsed, raw in surviving:
        tailored = await tailor(router, resume, parsed)
        resume_text = resume_to_plain_text(tailored.tailored_resume)
        v = validate_tailored(
            resume_text=resume_text,
            must_haves=parsed.get("must_haves", []),
            cover_letter=tailored.cover_letter,
        )
        if not v.ok:
            match.status = "tailoring_failed"
            match.reasoning = {"validation_issues": v.issues}
            continue

        # Persist artifacts to object store
        resume_pdf = render_resume_pdf(tailored.tailored_resume)
        cl_pdf = render_cover_letter_pdf(tailored.cover_letter)
        resume_key = f"tailored/{user.id}/{match.id}/resume.pdf"
        cl_key = f"tailored/{user.id}/{match.id}/cover_letter.pdf"
        put_object(resume_key, resume_pdf, "application/pdf")
        put_object(cl_key, cl_pdf, "application/pdf")

        ta = TailoredArtifact(
            match_id=match.id, resume_pdf_path=resume_key,
            resume_docx_path=None, resume_text=resume_text,
            cover_letter_pdf=cl_key, cover_letter_text=tailored.cover_letter,
            model_used=tailored.model_used, token_cost_usd=tailored.cost_usd,
        )
        db.add(ta)
        match.status = "tailored"
    db.commit()
    return [m for m, _, _ in surviving]


def _candidate_level(resume: dict) -> str:
    """Best-effort: use the most recent role's seniority."""
    if not resume.get("experience"):
        return "unknown"
    title = resume["experience"][0].get("title", "").lower()
    if "vp" in title or "director" in title:
        return "director"
    if "principal" in title or "staff" in title:
        return "principal"
    if "manager" in title or "head" in title:
        return "manager"
    if "senior" in title or "sr" in title:
        return "IC4"
    if "junior" in title or "jr" in title:
        return "IC3"
    return "IC3"


def _comp_match(comp: dict | None, prefs: dict) -> float:
    if not comp or (comp.get("min") is None and comp.get("max") is None):
        return 0.5  # not stated, benefit of doubt
    pref_min = prefs.get("salary_min")
    if pref_min is None:
        return 0.8
    lo = comp.get("min") or comp.get("max") or 0
    if lo >= pref_min:
        return 1.0
    if lo >= pref_min * 0.85:
        return 0.5
    return 0.0


def _upsert_match(
    db: Session, user_id: uuid.UUID, job_id: uuid.UUID,
    prob: float, jd_q: float, sen: float, mh: float, comp: float,
) -> Match:
    stmt = pg_insert(Match).values(
        user_id=user_id, job_id=job_id, match_score=prob,
        jd_quality=jd_q, seniority_match=sen, must_have_cov=mh, comp_match=comp,
    ).on_conflict_do_update(
        index_elements=["user_id", "job_id"],
        set_={"match_score": prob, "jd_quality": jd_q, "seniority_match": sen,
              "must_have_cov": mh, "comp_match": comp},
    ).returning(Match)
    return db.execute(stmt).scalar_one()
```

```python
    if file.content_type not in ("application/pdf", "application/vnd.openxmlformats-officedocument.wordprocessingml.document", "text/plain"):
        raise HTTPException(status.HTTP_415_UNSUPPORTED_MEDIA_TYPE)
    data = await file.read()
    text = extract_text(data, file.filename or "resume.pdf")
    p = db.get(Profile, user.id)
    if not p:
        raise HTTPException(status.HTTP_404_NOT_FOUND, "no profile")
    # Persist raw file
    ext = (file.filename or "resume").split(".")[-1]
    key = f"resumes/{user.id}/original.{ext}"
    put_object(key, data, file.content_type)
    if ext == "pdf":
        p.resume_pdf_path = key
    elif ext == "docx":
        p.resume_docx_path = key
    # Structure via LLM
    from app.core.config import get_settings
    from app.llm.opencode_zen import OpenCodeZenProvider
    router = LLMRouter({"opencode_zen": OpenCodeZenProvider()})
    p.resume_parsed_json = await structure_resume(router, text)
    db.commit()
    db.refresh(p)
    return p
```

```python
@router.post("/matches/{match_id}/apply/assisted")
def assisted_apply(
    match_id: str,
    user: User = Depends(current_user),
    db: Session = Depends(get_db),
) -> dict:
    m = db.get(Match, match_id)
    if not m or m.user_id != user.id:

        resume_snapshot=ta.resume_pdf_path, cover_letter_snap=ta.cover_letter_pdf,
    )
    db.add(app_row)
    m.status = "queued_apply"
    db.commit()
    return {
        "status": "ok",
        "job_url": job.url,
        "resume_pdf_path": ta.resume_pdf_path,
        "cover_letter_pdf_path": ta.cover_letter_pdf,
```

```python
import asyncio
import logging

from apscheduler.schedulers.asyncio import AsyncIOScheduler
from sqlalchemy import create_engine, select
from sqlalchemy.orm import sessionmaker

from app.core.config import get_settings
from app.db.models import Profile, User
from app.llm.opencode_zen import OpenCodeZenProvider
from app.llm.router import LLMRouter
from app.pipeline.run import run_for_user
from app.sources.greenhouse import GreenhouseSource

log = logging.getLogger(__name__)


def build_router() -> LLMRouter:
    return LLMRouter({"opencode_zen": OpenCodeZenProvider()})


async def refresh_all_users() -> None:
    """One-shot refresh: run pipeline for all users with profiles."""
    s = get_settings()
    engine = create_engine(s.database_url, future=True)
    Session = sessionmaker(bind=engine)
    router = build_router()

    with Session() as db:
        users = db.scalars(select(User)).all()
    for u in users:
        with Session() as db:
            prof = db.get(Profile, u.id)
            if not prof or not prof.resume_parsed_json:
                log.info("skip %s — no profile or resume", u.email)
                continue
            src = GreenhouseSource(board_token="acme")  # TODO: read from per-profile config
            raw_jobs = await src.fetch()
            await run_for_user(db, u, prof, raw_jobs, router)
        log.info("done %s", u.email)


def start_scheduler() -> AsyncIOScheduler:
    sched = AsyncIOScheduler()
    sched.add_job(refresh_all_users, "cron", hour=8, minute=0, id="daily_refresh")
    sched.start()
    return sched
```

```python
from sqlalchemy import select

    p.add_argument("--display-name", required=True)
    p.add_argument("--password", required=True)
    p.add_argument("--role", default="member")

        if existing:
            print(f"user {args.email} already exists, updating password")
            existing.password_hash = hash_password(args.password)
            existing.role = args.role
            user = existing
        else:
            user = User(
                email=args.email,
                display_name=args.display_name,
                role=args.role,
                password_hash=hash_password(args.password),

            db.add(user)
            db.flush()
        prof = db.get(Profile, user.id)
        if prof is None:
            db.add(Profile(
                user_id=user.id,
                market_primary=args.market,
                llm_default_model=args.llm_model,
            ))
        print(f"OK: {user.email} (role={user.role})")


if __name__ == "__main__":
```
